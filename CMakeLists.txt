cmake_minimum_required(VERSION 3.16)
project(nnn_threadpool VERSION 1.0 LANGUAGES CXX)

find_package(Threads REQUIRED)
add_library(nnn_threadpool INTERFACE)
add_library(nnn::threadpool ALIAS nnn_threadpool)
target_include_directories(nnn_threadpool INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_features(nnn_threadpool INTERFACE cxx_std_17)
target_link_libraries(nnn_threadpool INTERFACE Threads::Threads)

include(CTest)
option(NNN_THREADPOOL_SANITIZE "Enable address and undefined-behavior sanitizers" OFF)
option(NNN_THREADPOOL_TSAN "Enable ThreadSanitizer (requires a supported platform)" OFF)
option(NNN_THREADPOOL_BUILD_EXAMPLES "Build the standalone usage example" OFF)
if(NNN_THREADPOOL_SANITIZE AND NNN_THREADPOOL_TSAN)
    message(FATAL_ERROR "Choose either ASan/UBSan or TSan, not both")
endif()
if(BUILD_TESTING)
    add_subdirectory(test)
endif()
if(NNN_THREADPOOL_BUILD_EXAMPLES)
    add_executable(threadpool_example examples/basic.cpp)
    target_link_libraries(threadpool_example PRIVATE nnn::threadpool)
endif()
