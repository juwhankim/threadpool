add_executable(threadpool_tests correctness.cpp)
add_executable(threadpool_benchmark benchmark.cpp)
set(threadpool_test_targets threadpool_tests threadpool_benchmark)
# The text workload is an optional integration test with the sibling package;
# the vendored threadpool itself remains independent of normalization.
set(normalization_root "${CMAKE_CURRENT_SOURCE_DIR}/../../basic_normalization")
if(EXISTS "${normalization_root}/src/unicode_normalization.cpp")
    add_executable(threadpool_text_processing text_processing.cpp "${normalization_root}/src/unicode_normalization.cpp")
    target_include_directories(threadpool_text_processing PRIVATE "${normalization_root}/include")
    list(APPEND threadpool_test_targets threadpool_text_processing)
    add_test(NAME threadpool_text_processing COMMAND threadpool_text_processing --test)
    set_tests_properties(threadpool_text_processing PROPERTIES TIMEOUT 60)
endif()
foreach(target IN LISTS threadpool_test_targets)
    target_link_libraries(${target} PRIVATE nnn_threadpool)
    set_target_properties(${target} PROPERTIES CXX_EXTENSIONS OFF)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX)
    elseif(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
        if(NNN_THREADPOOL_SANITIZE)
            target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
            target_link_options(${target} PRIVATE -fsanitize=address,undefined)
        elseif(NNN_THREADPOOL_TSAN)
            target_compile_options(${target} PRIVATE -fsanitize=thread -fno-omit-frame-pointer)
            target_link_options(${target} PRIVATE -fsanitize=thread)
        endif()
    endif()
endforeach()
add_test(NAME threadpool_correctness COMMAND threadpool_tests)
set_tests_properties(threadpool_correctness PROPERTIES TIMEOUT 60)
